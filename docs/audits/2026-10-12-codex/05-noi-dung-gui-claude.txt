43: async function splitPdf(buffer, pages) {
44:   if (pages <= PDF_CHUNK_PAGES && buffer.length <= PDF_CHUNK_BYTES) return [{ buffer, from: 1, to: pages }];
45:   const src = await PDFDocument.load(buffer, { ignoreEncryption: true, updateMetadata: false });
46:   const perChunk = Math.max(1, Math.min(PDF_CHUNK_PAGES, Math.floor((pages * PDF_CHUNK_BYTES) / buffer.length)));
47:   const out = [];
48:   async function build(from, to) {
49:     const doc = await PDFDocument.create();
50:     const idx = Array.from({ length: to - from + 1 }, (_, i) => from - 1 + i);
51:     for (const p of await doc.copyPages(src, idx)) doc.addPage(p);
52:     const bytes = Buffer.from(await doc.save());
53:     if (bytes.length > PDF_CHUNK_BYTES && to > from) {
54:       const mid = Math.floor((from + to) / 2);
55:       await build(from, mid);
56:       await build(mid + 1, to);
57:     } else {
58:       out.push({ buffer: bytes, from, to });
59:     }
60:   }
61:   for (let from = 1; from <= pages; from += perChunk) await build(from, Math.min(pages, from + perChunk - 1));
62:   return out;
63: }
